cmake_minimum_required(VERSION 3.8)
project(agt_map_processing)
find_package(ament_cmake REQUIRED)
set(CMAKE_CXX_STANDARD 17)
find_library(OCTOMAP_LIBRARY octomap REQUIRED)
find_library(OCTOMATH_LIBRARY octomath REQUIRED)
add_executable(octomap_builder scripts/build_octomap.cpp)
target_link_libraries(octomap_builder ${OCTOMAP_LIBRARY} ${OCTOMATH_LIBRARY})
install(TARGETS octomap_builder DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/run_pipeline.py DESTINATION lib/${PROJECT_NAME} RENAME process_map)
install(PROGRAMS scripts/prepare_map.py DESTINATION lib/${PROJECT_NAME} RENAME prepare_map)
install(FILES scripts/centerpoint_postprocess.py scripts/ground_grid.py
  scripts/centerpoint_config.json scripts/prepare_map.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY algorithms DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME test_algorithm_pipeline COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_algorithm_pipeline.py)
endif()
ament_package()
